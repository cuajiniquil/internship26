benchmarkSys <- function(trset, fn, times) {
  ttaken <- rep(0, times)
  #warmup?
  fn(trset[[1]])
  
  for (t in 1:times) {
    tstart <- Sys.time()
    fn(trset[[t]])
    tend <- Sys.time()
    ttaken[t] <- (tend - tstart) * 10^6
  }
  return(ttaken)
}

load_newick_trees <- function(file_path) {
  lines <- readLines(file_path)
  lines <- lines[nzchar(lines)]
  lapply(lines, function(line) read_tree(string = line))
}

# interval extraction - native postorder 
extract_coalescent_intervals <- function(tree) {
  n_tips  <- length(tree$tip.label)
  n_nodes <- n_tips + tree$Nnode
  
  # native root distances: atomic vector indexed by node number
  dists <- castor::get_all_distances_to_root(tree, as_edge_count = FALSE)
  dists <- dists[seq_len(n_nodes)]
  
  present <- max(dists[seq_len(n_tips)])
  ages    <- present - dists
  
  events <- data.frame(
    time = ages,
    type = c(rep("sample", n_tips), rep("coalescent", tree$Nnode)),
    name = c(tree$tip.label, as.character((n_tips + 1):n_nodes)),
    stringsAsFactors = FALSE
  )
  events <- events[order(events$time), ]
  rownames(events) <- NULL
  
  n_active  <- 0
  intervals <- list()
  for (i in seq_len(nrow(events) - 1)) {
    n_active <- n_active + ifelse(events$type[i] == "sample", 1L, -1L)
    intervals[[i]] <- list(
      a         = events$time[i],
      b         = events$time[i + 1],
      k         = n_active,
      ends_coal = events$type[i + 1] == "coalescent"
    )
  }
  
  intervals
}

# ------------------------------------------------------------------
# Log-likelihood of a coalescent tree under exponential growth.
#
# intervals : data.frame with columns t0, t1, k
#   t0 : start time of interval (older, closer to root)
#   t1 : end time of interval   (younger, closer to present)
#   k  : number of active lineages in the interval
# ------------------------------------------------------------------
FIXED_PHI0 <- 1.0
FIXED_R    <- 0.023882220114924


coalescent_loglikelihood <- function(intervals,
                                     phi0 = FIXED_PHI0,
                                     r    = FIXED_R) {
  log_phi0 <- log(phi0)
  logL     <- 0.0
  
  for (iv in intervals) {
    k  <- iv$k
    a  <- iv$a
    b  <- iv$b
    c2 <- k * (k - 1) / 2
    
    if (c2 <= 0) next

    # survival term: coalescent rate λ(t) = (1/φ0) * exp(r*t)    
    if (r == 0) {
      logL <- logL - c2 * (b - a) / phi0
    } else {
      logL <- logL - (c2 / (phi0 * r)) * (exp(r * b) - exp(r * a))
    }
    
    # event term only when the interval ends in a coalescence
    if (iv$ends_coal) {
      logL <- logL + log(c2) - log_phi0 + r * b
    }
  }
  
  logL
}


# Public wrapper: extracts intervals from the castor tree, then
# computes the coalescent log-likelihood.
loglik_castor <- function(tree,
                          phi0 = FIXED_PHI0,
                          r    = FIXED_R) {
  intervals <- extract_coalescent_intervals(tree)
  coalescent_loglikelihood(intervals, phi0, r)
}

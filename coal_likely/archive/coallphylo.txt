function is_ultrametric(tree::AbstractTree)::Bool
    """
    Check if all leaf nodes are equidistant from root.
    """
    function get_leaf_distances(node, dist=0.0)
        if isleaf(tree, node)
            return [dist]
        end
        distances = Float64[]
        for child in getchildren(tree, node)
            branch = getinbound(tree, child)
            child_dist = dist + getlength(tree, branch)
            append!(distances, get_leaf_distances(child, child_dist))
        end
        return distances
    end

    distances = get_leaf_distances(getroot(tree))
    if length(distances) <= 1
        return true
    end
    # Check if all distances are approximately equal
    return maximum(distances) - minimum(distances) < 1e-10
end

#replace with @benchmark 
function benchmark_sys(tree_set, fn::Function, times::Int)::Vector{Float64}
    """Benchmark function execution time over a list of trees."""
    ttaken = Float64[]

    fn(tree_set[1])

    for t in 1:times
        tree = tree_set[mod1(t, length(tree_set))]
        tstart = time_ns()
        fn(tree)
        tend = time_ns()

        elapsed_us = (tend - tstart) / 1000.0
        push!(ttaken, elapsed_us)
    end

    return ttaken
end

function traverse_tree_preorder_man(tree::AbstractTree)::Int
    count = 0
    stack = [getroot(tree)]
    while !isempty(stack)
        node = pop!(stack)
        count += 1
        for child in getchildren(tree, node)
            push!(stack, child)
        end
    end
    return count
end


function traverse_tree_postorder_man(tree::AbstractTree)::Int
    """Traverse all nodes in tree (postorder)."""
    count = 0
    stack = [getroot(tree)]
    visited = Set()

    while !isempty(stack)
        node = last(stack)
        if node in visited
            pop!(stack)
            count += 1
        else
            push!(visited, node)
            for child in getchildren(tree, node)
                push!(stack, child)
            end
        end
    end
    return count
end


function traverse_tree_levelorder_man(tree::AbstractTree)::Int
    """Traverse all nodes in tree (levelorder)."""
    count = 0
    queue = [getroot(tree)]

    while !isempty(queue)
        node = popfirst!(queue)
        count += 1
        for child in getchildren(tree, node)
            push!(queue, child)
        end
    end
    return count
end


function traverse_tree_preorder(tree::AbstractTree)::Int
    count = 0
    for _ in traversal(tree, preorder)
        count += 1
    end
    return count
end


function traverse_tree_postorder(tree::AbstractTree)::Int
    count = 0
    for _ in traversal(tree, postorder)
        count += 1
    end
    return count
end


function traverse_tree_levelorder(tree::AbstractTree)::Int
    count = 0
    for _ in traversal(tree, breadthfirst)
        count += 1
    end
    return count
end

function _build_intervals(tree::AbstractTree, depths::Dict)
    """
    Build coalescent intervals from node depths.
    Shared helper used by both preorder and levelorder extractors.
    """
    leaves   = filter(n -> isleaf(tree, n), collect(keys(depths)))
    internal = filter(n -> !isleaf(tree, n), collect(keys(depths)))

    present = maximum(depths[l] for l in leaves)
    ages    = Dict(n => present - d for (n, d) in depths)

    events = vcat(
        [(time=ages[l], type="sample",     name=string(l)) for l in leaves],
        [(time=ages[n], type="coalescent", name=string(n)) for n in internal]
    )
    sort!(events, by=e -> e.time)

    intervals = []
    n_active  = 0
    for i in 1:length(events)-1
        n_active += events[i].type == "sample" ? 1 : -1
        push!(intervals, (
            a         = events[i].time,
            b         = events[i+1].time,
            k         = n_active,
            ends_coal = events[i+1].type == "coalescent"
        ))
    end

    return intervals
end


function extract_coalescent_intervals_preorder(tree::AbstractTree)
    """
    Extract coalescent intervals using a manual preorder traversal.
    Reuses the same stack pattern as traverse_tree_preorder_man.
    """
    root   = getroot(tree)
    depths = Dict{typeof(root), Float64}(root => 0.0)
    stack  = [root]

    while !isempty(stack)
        node = pop!(stack)
        for child in getchildren(tree, node)
            branch = getinbound(tree, child)
            bl     = branch === nothing ? 0.0 : getlength(tree, branch)
            depths[child] = depths[node] + bl
            push!(stack, child)
        end
    end

    return _build_intervals(tree, depths)
end


function extract_coalescent_intervals_levelorder(tree::AbstractTree)
    """
    Extract coalescent intervals using a manual levelorder traversal.
    Reuses the same queue pattern as traverse_tree_levelorder_man.
    """
    root   = getroot(tree)
    depths = Dict{typeof(root), Float64}(root => 0.0)
    queue  = [root]

    while !isempty(queue)
        node = popfirst!(queue)
        for child in getchildren(tree, node)
            branch = getinbound(tree, child)
            bl     = branch === nothing ? 0.0 : getlength(tree, branch)
            depths[child] = depths[node] + bl
            push!(queue, child)
        end
    end

    return _build_intervals(tree, depths)
end


const FIXED_PHI0 = 1.0
const FIXED_R    = 0.023882220114924

function coalescent_loglikelihood(intervals)
    logL = 0.0
    log_phi0 = log(FIXED_PHI0)
    for iv in intervals
        k, a, b, ends = iv.k, iv.a, iv.b, iv.ends_coal
        c2 = k * (k - 1) / 2
        c2 <= 0 && continue

        if FIXED_R == 0
            logL -= c2 * (b - a) / FIXED_PHI0
        else
            logL -= (c2 / (FIXED_PHI0 * FIXED_R)) * (
                exp(FIXED_R * b) - exp(FIXED_R * a)
            )
        end

        if ends
            logL += log(c2) - log_phi0 + FIXED_R * b
        end
    end
    logL
end


loglik_phylo(tree) = coalescent_loglikelihood(extract_coalescent_intervals(tree))

function load_newick_trees(file_path::String)
    trees = Any[]
    for line in readlines(file_path)
        stripped = strip(line)
        if !isempty(stripped)
            push!(trees, parsenewick(String(stripped)))
        end
    end
    return trees
end

function print_stats(tree_label, results)
    println("\n=== Phylo.jl Benchmark Statistics ($tree_label) ===")
    for method_name in ["preorder", "postorder", "levelorder", "is_ultrametric"]
        method_times = results[method_name]
        println("\n$method_name:")
        println("  Mean time (µs): $(round(mean(method_times), digits=4))")
        println("  Median time (µs): $(round(median(method_times), digits=4))")
        println("  Std dev (µs): $(round(std(method_times), digits=4))")
        println("  Min time (µs): $(round(minimum(method_times), digits=4))")
        println("  Max time (µs): $(round(maximum(method_times), digits=4))")
    end
end

function save_plot(results, title::String, filename::String)
    fig, ax = subplots(figsize=(10, 6))
    data_by_method = [results["preorder"], results["postorder"], results["levelorder"], results["is_ultrametric"]]
    positions = [1, 2, 3, 4]
    ax.violinplot(data_by_method, positions=positions, widths=0.7)
    ax.set_ylabel("Time (microseconds)")
    ax.set_title(title)
    ax.set_xticks(positions)
    ax.set_xticklabels(["preorder", "postorder", "levelorder", "is_ultrametric"])
    ax.grid(true, axis="y", alpha=0.3)

    tight_layout()
    savefig(filename, dpi=100)
    println("\nPlot saved to '$filename'")
    close()
end
